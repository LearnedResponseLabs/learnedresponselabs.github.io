Reading editions

A reading edition sets each statement of a paper beside the frozen Lean statement that checks it, or the source it rests on, so you can see exactly what the machine verified and what it did not. Every edition is reviewed internally before its paper is released; external reviews are noted as they arrive.

The paper · Corollary 31 · p. 39

Corollary 31 (inverting Gf is minimizing). For every y ∈ Gf(Θf>0) there is exactly one θ ∈ Θf>0 with Gf(θ) = y, and it is the unique minimizer of Fy = Ψf − ⟨y, ·⟩ on Θf>0.

theorem variational_gradient_preimage (F : LooplessBAP n) (a : Fin n → ℝ) :
    let δ := (fun (θ : Fin n → ℝ) (e : Fin n × Fin n) => θ e.2 - θ e.1 + Real.pi * (if e.2 < e.1 then 1 else 0))
    let Ψ := (fun θ : Fin n → ℝ => ∑ j : EdgeIdx F, ∫ t in (Real.pi / 2)..(δ θ (edgeOf F j)), Real.log (Real.sin t))
    let G := (fun θ : Fin n → ℝ => (Matrix.of (fun (r : Fin n) (j : EdgeIdx F) => (Bmat F r j : ℝ))).mulVec (fun j => Real.log (Real.sin (δ θ (edgeOf F j)))))
    let S := {θ : Fin n → ℝ | (∀ p q, CrossEdge F p q → 0 < δ θ (p,q) ∧ δ θ (p,q) < Real.pi) ∧ ∀ p, (∀ q, Relation.ReflTransGen (fun p q => CrossEdge F p q ∨ CrossEdge F q p) p q → p ≤ q) → θ p = a p}
